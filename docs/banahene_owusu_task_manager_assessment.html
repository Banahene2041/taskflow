1. Overview & design decisions

TaskFlow is a guest-session Kanban board inspired by Asana and Linear: calm surfaces, clear hierarchy, and intentional interaction design. The stack is Next.js 16 (App Router), TypeScript, Tailwind CSS 4, shadcn/ui, Supabase (anonymous auth + Postgres RLS), and @dnd-kit for drag-and-drop.

Design tokens live in CSS variables for light and dark themes (Asana-like neutrals, coral Create button, blue active tabs). The shell uses a persistent sidebar, sticky header with search, List and Board views, and a horizontally scrollable board so columns stay readable on smaller screens.

Optimistic drag-and-drop keeps status and order updates snappy; writes persist to Supabase on drop. Each browser gets an anonymous guest session so data is isolated by user_id under RLS.

2. Required features

  • Kanban columns: To Do, In Progress, In Review, Done
  • Drag tasks between columns (status updates on drop)
  • Create tasks with title + optional description, priority, due date, assignee, labels
  • Guest anonymous auth with automatic session on first launch
  • Row Level Security so each guest only reads/writes their own data
  • Loading, empty, and error states
  • Responsive layout (sidebar sheet on mobile)
  • Hosted live demo on Vercel

3. Advanced features built

  1. Team members & assignees — create members with color, assign on create/edit, avatars on cards
  2. Task comments — detail panel, chronological timestamps, comments table
  3. Activity log — trigger-written timeline (created, status moves, edits, assignments)
  4. Labels / tags — create labels, multi-assign, filter board by label
  5. Due date indicators — overdue / due-soon styling on cards
  6. Search & filtering — header search by title; toolbar filters for priority, assignee, label
  7. Board summary — total, completed, and overdue counts above the board

Extras: List view with the same DnD model; within-column reordering via position.

4. Local setup

  1. pnpm install
  2. Copy .env.example → .env.local with NEXT_PUBLIC_SUPABASE_URL and the publishable/anon key only
  3. Supabase: enable Authentication → Providers → Anonymous
  4. Run supabase/schema.sql in the SQL Editor
  5. pnpm dev → http://localhost:3000 (guest session starts automatically)

Security: Never commit the service-role / secret key. Only the public anon/publishable key belongs in the frontend.

5. Database schema summary

TablePurpose
tasksBoard items: id, title, status, user_id, created_at + description, priority, due_date, assignee_id, position
team_membersGuest-owned team for assignees
labels / task_labelsCustom tags + many-to-many
commentsTask discussion
task_activityHistory (trigger-backed)

RLS is enabled on all tables (auth.uid() ownership). Full SQL is in the appendix.

6. Tradeoffs & future work

  • Optimistic client board state (no Realtime yet) — drop persistence meets the brief
  • Anonymous guests cannot recover a session after clearing storage
  • Single assignee per task (multi-assignee needs a join table)
  • CAPTCHA / rate limits would harden anonymous sign-in in production
  • Richer descriptions and attachments as natural follow-ups

7. Security awareness

  • Frontend uses only NEXT_PUBLIC_SUPABASE_URL + publishable/anon key
  • RLS policies bind CRUD to auth.uid()
  • Service-role key is never used and is not committed

-- TaskFlow schema + RLS
-- Run this in the Supabase SQL Editor (Dashboard → SQL → New query).
-- Also enable Anonymous Sign-Ins: Authentication → Providers → Anonymous.

create extension if not exists "pgcrypto";

-- ---------------------------------------------------------------------------
-- Tables
-- ---------------------------------------------------------------------------

create table if not exists public.team_members (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users (id) on delete cascade,
  name text not null check (char_length(trim(name)) > 0),
  color text not null default '#635BFF',
  avatar_url text,
  created_at timestamptz not null default now()
);

create table if not exists public.labels (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users (id) on delete cascade,
  name text not null check (char_length(trim(name)) > 0),
  color text not null default '#635BFF',
  created_at timestamptz not null default now(),
  unique (user_id, name)
);

create table if not exists public.tasks (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users (id) on delete cascade,
  title text not null check (char_length(trim(title)) > 0),
  description text,
  status text not null default 'todo'
    check (status in ('todo', 'in_progress', 'in_review', 'done')),
  priority text not null default 'normal'
    check (priority in ('low', 'normal', 'high')),
  due_date date,
  assignee_id uuid references public.team_members (id) on delete set null,
  position integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- Existing projects: add position if the table was created before this column existed.
alter table public.tasks
  add column if not exists position integer not null default 0;

-- Backfill stable per-status positions for rows that still share the default.
with ranked as (
  select
    id,
    row_number() over (
      partition by user_id, status
      order by position asc, created_at asc
    ) - 1 as next_position
  from public.tasks
)
update public.tasks as t
set position = ranked.next_position
from ranked
where t.id = ranked.id
  and t.position is distinct from ranked.next_position;

create table if not exists public.task_labels (
  task_id uuid not null references public.tasks (id) on delete cascade,
  label_id uuid not null references public.labels (id) on delete cascade,
  primary key (task_id, label_id)
);

create table if not exists public.comments (
  id uuid primary key default gen_random_uuid(),
  task_id uuid not null references public.tasks (id) on delete cascade,
  user_id uuid not null references auth.users (id) on delete cascade,
  body text not null check (char_length(trim(body)) > 0),
  created_at timestamptz not null default now()
);

create table if not exists public.task_activity (
  id uuid primary key default gen_random_uuid(),
  task_id uuid not null references public.tasks (id) on delete cascade,
  user_id uuid not null references auth.users (id) on delete cascade,
  action text not null,
  meta jsonb not null default '{}'::jsonb,
  created_at timestamptz not null default now()
);

create index if not exists tasks_user_id_idx on public.tasks (user_id);
create index if not exists tasks_status_idx on public.tasks (user_id, status);
create index if not exists tasks_status_position_idx on public.tasks (user_id, status, position);
create index if not exists team_members_user_id_idx on public.team_members (user_id);
create index if not exists labels_user_id_idx on public.labels (user_id);
create index if not exists comments_task_id_idx on public.comments (task_id);
create index if not exists task_activity_task_id_idx on public.task_activity (task_id);

-- ---------------------------------------------------------------------------
-- updated_at trigger
-- ---------------------------------------------------------------------------

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists tasks_set_updated_at on public.tasks;
create trigger tasks_set_updated_at
before update on public.tasks
for each row
execute function public.set_updated_at();

-- ---------------------------------------------------------------------------
-- Activity helpers
-- ---------------------------------------------------------------------------

create or replace function public.log_task_activity()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if tg_op = 'INSERT' then
    insert into public.task_activity (task_id, user_id, action, meta)
    values (
      new.id,
      new.user_id,
      'created',
      jsonb_build_object('title', new.title, 'status', new.status)
    );
    return new;
  end if;

  if tg_op = 'UPDATE' then
    if old.status is distinct from new.status then
      insert into public.task_activity (task_id, user_id, action, meta)
      values (
        new.id,
        new.user_id,
        'status_changed',
        jsonb_build_object('from', old.status, 'to', new.status)
      );
    end if;

    if old.title is distinct from new.title
      or old.description is distinct from new.description
      or old.priority is distinct from new.priority
      or old.due_date is distinct from new.due_date then
      insert into public.task_activity (task_id, user_id, action, meta)
      values (
        new.id,
        new.user_id,
        'updated',
        jsonb_build_object(
          'title_changed', old.title is distinct from new.title,
          'priority_changed', old.priority is distinct from new.priority,
          'due_date_changed', old.due_date is distinct from new.due_date
        )
      );
    end if;

    if old.assignee_id is distinct from new.assignee_id then
      insert into public.task_activity (task_id, user_id, action, meta)
      values (
        new.id,
        new.user_id,
        'assignee_changed',
        jsonb_build_object('from', old.assignee_id, 'to', new.assignee_id)
      );
    end if;

    return new;
  end if;

  return new;
end;
$$;

drop trigger if exists tasks_log_activity on public.tasks;
create trigger tasks_log_activity
after insert or update on public.tasks
for each row
execute function public.log_task_activity();

  on public.team_members for select

drop policy if exists "task_labels_select_own" on public.task_labels;
create policy "task_labels_select_own"
  on public.task_labels for select
  to authenticated
  using (
    exists (
      select 1 from public.tasks t
      where t.id = task_id and t.user_id = auth.uid()
    )
  );

drop policy if exists "task_labels_insert_own" on public.task_labels;
create policy "task_labels_insert_own"
  on public.task_labels for insert
  to authenticated
  with check (
    exists (
      select 1 from public.tasks t
      where t.id = task_id and t.user_id = auth.uid()
    )
    and exists (
      select 1 from public.labels l
      where l.id = label_id and l.user_id = auth.uid()
    )
  );

drop policy if exists "task_labels_delete_own" on public.task_labels;
create policy "task_labels_delete_own"
  on public.task_labels for delete
  to authenticated
  using (
    exists (
      select 1 from public.tasks t
      where t.id = task_id and t.user_id = auth.uid()
    )
  );
